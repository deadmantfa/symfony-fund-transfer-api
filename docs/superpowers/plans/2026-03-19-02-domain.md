# Fund Transfer API — Plan 2: Domain Layer

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the pure PHP domain layer — entities, value objects, enums, and repository interfaces — with zero Symfony or Doctrine dependencies.

**Architecture:** Domain layer is the innermost ring. All classes are pure PHP 8.5. No framework imports. Entities own their invariants and enforce them via methods. Value objects are `readonly`. Enums replace all string/int constants. Repository interfaces define the contract — infrastructure implements them in Plan 3.

**Tech Stack:** PHP 8.5, brick/money (BigDecimal), Pest (unit tests)

```php
<?php

declare(strict_types=1);

namespace App\Domain\Shared\ValueObject;

use Brick\Math\BigDecimal;
use Brick\Math\RoundingMode;

readonly final class Money
{
    public readonly BigDecimal $amount;

    public function __construct(
        string|BigDecimal $amount,
        public readonly CurrencyCode $currency,
    ) {
        $this->amount = $amount instanceof BigDecimal
            ? $amount
            : BigDecimal::of($amount)->toScale(8, RoundingMode::UNNECESSARY);
    }

    public function add(self $other): self
    {
        $this->assertSameCurrency($other);

        return new self($this->amount->plus($other->amount), $this->currency);
    }

    public function subtract(self $other): self
    {
        $this->assertSameCurrency($other);

        return new self($this->amount->minus($other->amount), $this->currency);
    }

    public function multiply(string|BigDecimal $rate): self
    {
        $result = $this->amount->multipliedBy($rate)->toScale(8, RoundingMode::HALF_UP);

        return new self($result, $this->currency);
    }

    public function isPositive(): bool
    {
        return $this->amount->isGreaterThan(BigDecimal::zero());
    }

    public function isZero(): bool
    {
        return $this->amount->isEqualTo(BigDecimal::zero());
    }

    public function isSufficientFor(self $other): bool
    {
        $this->assertSameCurrency($other);

        return $this->amount->isGreaterThanOrEqualTo($other->amount);
    }

    public function equals(self $other): bool
    {
        return $this->currency->equals($other->currency)
            && $this->amount->isEqualTo($other->amount);
    }

    private function assertSameCurrency(self $other): void
    {
        if (!$this->currency->equals($other->currency)) {
            throw new \InvalidArgumentException(
                "Currency mismatch: {$this->currency} vs {$other->currency}"
            );
        }
    }

    public function __toString(): string
    {
        return $this->amount->__toString();
    }
}
```

```php
declare(strict_types=1);

namespace App\Domain\Shared\ValueObject;

readonly final class Uuid
{
    public function __construct(public readonly string $value)
    {
        if (!preg_match(
            '/^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i',
            $value
        )) {
            throw new \InvalidArgumentException("Invalid UUID: {$value}");
        }
    }

    public static function generate(): self
    {
        return new self(sprintf(
            '%04x%04x-%04x-%04x-%04x-%04x%04x%04x',
            mt_rand(0, 0xffff), mt_rand(0, 0xffff),
            mt_rand(0, 0xffff),
            mt_rand(0, 0x0fff) | 0x4000,
            mt_rand(0, 0x3fff) | 0x8000,
            mt_rand(0, 0xffff), mt_rand(0, 0xffff), mt_rand(0, 0xffff)
        ));
    }

    public function equals(self $other): bool
    {
        return strtolower($this->value) === strtolower($other->value);
    }

    public function __toString(): string
    {
        return $this->value;
    }
}
```

```php
declare(strict_types=1);

namespace App\Domain\Transfer;

enum TransferStatus: string
{
    case Pending    = 'pending';
    case Processing = 'processing';
    case Completed  = 'completed';
    case Failed     = 'failed';

    public function isFinal(): bool
    {
        return match ($this) {
            self::Completed, self::Failed => true,
            default                       => false,
        };
    }

    public function canTransitionTo(self $next): bool
    {
        return match ($this) {
            self::Pending    => $next === self::Processing || $next === self::Failed,
            self::Processing => $next === self::Completed || $next === self::Failed,
            default          => false,
        };
    }
}
```

```php
<?php

declare(strict_types=1);

namespace App\Domain\User;

use DateTimeImmutable;

final class User
{
    private function __construct(
        public private(set) readonly UserId $id,
        public private(set) string $email,
        public private(set) string $firstName,
        public private(set) string $lastName,
        public private(set) string $passwordHash,
        public private(set) ?string $phoneNumber,
        public private(set) ?DateTimeImmutable $dateOfBirth,
        public private(set) ?array $address,
        public private(set) KycStatus $kycStatus,
        public private(set) UserStatus $status,
        public private(set) ?DateTimeImmutable $emailVerifiedAt,
        public private(set) ?DateTimeImmutable $termsAcceptedAt,
        public private(set) readonly DateTimeImmutable $createdAt,
        public private(set) DateTimeImmutable $updatedAt,
    ) {}

    public static function register(
        UserId $id,
        string $email,
        string $firstName,
        string $lastName,
        string $passwordHash,
        ?string $phoneNumber = null,
        ?DateTimeImmutable $dateOfBirth = null,
        ?array $address = null,
    ): self {
        $now = new DateTimeImmutable();

        return new self(
            id: $id,
            email: $email,
            firstName: $firstName,
            lastName: $lastName,
            passwordHash: $passwordHash,
            phoneNumber: $phoneNumber,
            dateOfBirth: $dateOfBirth,
            address: $address,
            kycStatus: KycStatus::Unverified,
            status: UserStatus::Active,
            emailVerifiedAt: null,
            termsAcceptedAt: null,
            createdAt: $now,
            updatedAt: $now,
        );
    }

    public function verifyEmail(): void
    {
        $this->emailVerifiedAt = new DateTimeImmutable();
        $this->updatedAt = new DateTimeImmutable();
    }

    public function approveKyc(): void
    {
        if (!$this->status->isActive()) {
            throw new \DomainException('Cannot approve KYC for inactive user.');
        }

        $this->kycStatus = KycStatus::Verified;
        $this->updatedAt = new DateTimeImmutable();
    }

    public function suspend(): void
    {
        $this->status = UserStatus::Suspended;
        $this->updatedAt = new DateTimeImmutable();
    }

    public function close(): void
    {
        $this->status = UserStatus::Closed;
        $this->updatedAt = new DateTimeImmutable();
    }

    public function updateProfile(
        ?string $firstName = null,
        ?string $lastName = null,
        ?string $phoneNumber = null,
        ?array $address = null,
    ): void {
        if ($firstName !== null) {
            $this->firstName = $firstName;
        }
        if ($lastName !== null) {
            $this->lastName = $lastName;
        }
        if ($phoneNumber !== null) {
            $this->phoneNumber = $phoneNumber;
        }
        if ($address !== null) {
            $this->address = $address;
        }

        $this->updatedAt = new DateTimeImmutable();
    }

    public function canInitiateTransfer(): bool
    {
        return $this->status->isActive()
            && $this->kycStatus->allowsTransfers()
            && $this->emailVerifiedAt !== null;
    }
}
```

```php
<?php

declare(strict_types=1);

namespace App\Domain\Account;

use App\Domain\Shared\ValueObject\CurrencyCode;
use App\Domain\Shared\ValueObject\Money;
use App\Domain\User\UserId;
use DateTimeImmutable;

final class Account
{
    private function __construct(
        public private(set) readonly AccountId $id,
        public private(set) readonly UserId $userId,
        public private(set) readonly CurrencyCode $currency,
        public private(set) string $name,
        public private(set) Money $balance,
        public private(set) AccountStatus $status,
        public private(set) readonly DateTimeImmutable $createdAt,
        public private(set) DateTimeImmutable $updatedAt,
    ) {}

    public static function open(
        AccountId $id,
        UserId $userId,
        CurrencyCode $currency,
        string $name,
    ): self {
        $now = new DateTimeImmutable();

        return new self(
            id: $id,
            userId: $userId,
            currency: $currency,
            name: $name,
            balance: new Money('0.00000000', $currency),
            status: AccountStatus::Active,
            createdAt: $now,
            updatedAt: $now,
        );
    }

    public function credit(Money $amount): void
    {
        $this->assertActive();
        $this->balance = $this->balance->add($amount);
        $this->updatedAt = new DateTimeImmutable();
    }

    public function debit(Money $amount): void
    {
        $this->assertActive();

        if (!$this->balance->isSufficientFor($amount)) {
            throw new \DomainException(
                "Insufficient funds: balance {$this->balance} < debit {$amount}"
            );
        }

        $this->balance = $this->balance->subtract($amount);
        $this->updatedAt = new DateTimeImmutable();
    }

    public function freeze(): void
    {
        $this->assertActive();
        $this->status = AccountStatus::Frozen;
        $this->updatedAt = new DateTimeImmutable();
    }

    public function unfreeze(): void
    {
        if ($this->status !== AccountStatus::Frozen) {
            throw new \DomainException('Account is not frozen.');
        }

        $this->status = AccountStatus::Active;
        $this->updatedAt = new DateTimeImmutable();
    }

    public function close(): void
    {
        if (!$this->balance->isZero()) {
            throw new \DomainException('Cannot close account with non-zero balance.');
        }

        $this->status = AccountStatus::Closed;

            throw new \DomainException(
                "Account {$this->id} is not active (status: {$this->status->value})."
```

```php
<?php

declare(strict_types=1);

namespace App\Domain\Transfer;

use App\Domain\Account\AccountId;
use App\Domain\Shared\ValueObject\CurrencyCode;
use App\Domain\Shared\ValueObject\Money;
use App\Domain\User\UserId;
use DateTimeImmutable;

final class Transfer
{
    private function __construct(
        public private(set) readonly TransferId $id,
        public private(set) readonly AccountId $fromAccountId,
        public private(set) readonly AccountId $toAccountId,
        public private(set) readonly Money $amount,
        public private(set) readonly string $idempotencyKey,
        public private(set) readonly UserId $initiatedByUserId,
        public private(set) readonly string $ipAddress,
        public private(set) readonly string $userAgent,
        public private(set) TransferStatus $status,
        public private(set) ?Money $fromAmount,
        public private(set) ?Money $toAmount,
        public private(set) ?string $fxRate,
        public private(set) ?string $fxRateProvider,
        public private(set) ?string $failureReason,
        public private(set) readonly DateTimeImmutable $createdAt,
        public private(set) DateTimeImmutable $updatedAt,
    ) {}

    public static function initiate(
        TransferId $id,
        AccountId $fromAccountId,
        AccountId $toAccountId,
        Money $amount,
        string $idempotencyKey,
        UserId $initiatedByUserId,
        string $ipAddress,
        string $userAgent,
    ): self {
        $now = new DateTimeImmutable();

        return new self(
            id: $id,
            fromAccountId: $fromAccountId,
            toAccountId: $toAccountId,
            amount: $amount,
            idempotencyKey: $idempotencyKey,
            initiatedByUserId: $initiatedByUserId,
            ipAddress: $ipAddress,
            userAgent: $userAgent,
            status: TransferStatus::Pending,
            fromAmount: null,
            toAmount: null,
            fxRate: null,
            fxRateProvider: null,
            failureReason: null,
            createdAt: $now,
            updatedAt: $now,
        );
    }

    public function markProcessing(): void
    {
        $this->assertCanTransitionTo(TransferStatus::Processing);
        $this->status = TransferStatus::Processing;
        $this->updatedAt = new DateTimeImmutable();
    }

    public function complete(
        Money $fromAmount,
        Money $toAmount,
        string $fxRate,
        string $fxRateProvider,
    ): void {
        $this->assertCanTransitionTo(TransferStatus::Completed);
        $this->status = TransferStatus::Completed;
        $this->fromAmount = $fromAmount;
        $this->toAmount = $toAmount;
        $this->fxRate = $fxRate;
        $this->fxRateProvider = $fxRateProvider;
        $this->updatedAt = new DateTimeImmutable();
    }

    public function fail(string $reason): void
    {
        $this->assertCanTransitionTo(TransferStatus::Failed);
        $this->status = TransferStatus::Failed;
        $this->failureReason = $reason;
        $this->updatedAt = new DateTimeImmutable();
    }

    public function cancel(): void
    {
        if ($this->status !== TransferStatus::Pending) {
            throw new \DomainException(
                "Only pending transfers can be cancelled (current: {$this->status->value})."
            );
        }

        $this->fail('cancelled_by_user');
    }

    private function assertCanTransitionTo(TransferStatus $next): void
    {
        if (!$this->status->canTransitionTo($next)) {
            throw new \DomainException(
                "Cannot transition transfer from {$this->status->value} to {$next->value}."
            );
        }
    }
}
```
