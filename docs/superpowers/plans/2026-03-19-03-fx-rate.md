# Fund Transfer API — Plan 3: FX Rate Service

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement the FX rate service — Frankfurter HTTP provider behind `FxRateProviderInterface`, Redis caching with DB fallback, and a stub provider for tests.

**Architecture:** `CachingFxRateService` wraps `FxRateProviderInterface`. Resolution chain: Redis cache → Frankfurter API → last-known rate from DB → `FxRateUnavailableException`. The stub provider returns deterministic rates for all tests without HTTP calls.

**Tech Stack:** PHP 8.5, Symfony HttpClient, Redis (Symfony Cache), Doctrine (for DB fallback), Pest

**Prerequisite:** Plans 1 and 2 complete.

```php
<?php

declare(strict_types=1);

namespace App\Infrastructure\Persistence;

use App\Domain\FxRate\FxRate;
use App\Domain\FxRate\FxRateId;
use App\Domain\FxRate\FxRateRepository;
use App\Domain\Shared\ValueObject\CurrencyCode;
use Brick\Math\BigDecimal;
use Doctrine\DBAL\Connection;

final class DoctrineFxRateRepository implements FxRateRepository
{
    public function __construct(private readonly Connection $connection) {}

    public function findLatest(CurrencyCode $from, CurrencyCode $to): ?FxRate
    {
        $row = $this->connection->fetchAssociative(
            'SELECT * FROM fx_rates WHERE base_currency = ? AND target_currency = ?
             ORDER BY fetched_at DESC LIMIT 1',
            [(string) $from, (string) $to]
        );

        if ($row === false) {
            return null;
        }

        return new FxRate(
            id: new FxRateId($row['id']),
            baseCurrency: new CurrencyCode($row['base_currency']),
            targetCurrency: new CurrencyCode($row['target_currency']),
            rate: BigDecimal::of($row['rate']),
            provider: $row['provider'],
            fetchedAt: new \DateTimeImmutable($row['fetched_at']),
        );
    }

    public function save(FxRate $rate): void
    {
        $this->connection->executeStatement(
            'INSERT INTO fx_rates (id, base_currency, target_currency, rate, provider, fetched_at)
             VALUES (?, ?, ?, ?, ?, ?)
             ON CONFLICT (base_currency, target_currency, provider)
             DO UPDATE SET rate = EXCLUDED.rate, fetched_at = EXCLUDED.fetched_at',
            [
                (string) $rate->id,
                (string) $rate->baseCurrency,
                (string) $rate->targetCurrency,
                (string) $rate->rate,
                $rate->provider,
                $rate->fetchedAt->format('Y-m-d H:i:s'),
            ]
        );
    }

    public function nextIdentity(): FxRateId
    {
        return FxRateId::generate();
    }
}
```

```php
<?php

declare(strict_types=1);

namespace App\Infrastructure\FxRate;

use App\Domain\FxRate\FxRate;
use App\Domain\FxRate\FxRateId;
use App\Domain\FxRate\FxRateProviderInterface;
use App\Domain\FxRate\FxRateUnavailableException;
use App\Domain\Shared\ValueObject\CurrencyCode;
use Brick\Math\BigDecimal;
use Symfony\Contracts\HttpClient\HttpClientInterface;

final class FrankfurterFxRateProvider implements FxRateProviderInterface
{
    public function __construct(
        private readonly HttpClientInterface $httpClient,
        private readonly string $baseUrl,
    ) {}

    public function getRate(CurrencyCode $from, CurrencyCode $to): FxRate
    {
        try {
            $response = $this->httpClient->request(
                'GET',
                "{$this->baseUrl}/latest",
                ['query' => ['from' => (string) $from, 'to' => (string) $to]]
            );

            $data = $response->toArray();

            if (!isset($data['rates'][(string) $to])) {
                throw new FxRateUnavailableException($from, $to);
            }

            return new FxRate(
                id: FxRateId::generate(),
                baseCurrency: $from,
                targetCurrency: $to,
                rate: BigDecimal::of((string) $data['rates'][(string) $to])->toScale(10),
                provider: 'frankfurter',
                fetchedAt: new \DateTimeImmutable(),
            );
        } catch (FxRateUnavailableException $e) {
            throw $e;
        } catch (\Throwable $e) {
            throw new FxRateUnavailableException($from, $to);
        }
    }

    public function getSupportedCurrencies(): array
    {
        try {
            $response = $this->httpClient->request('GET', "{$this->baseUrl}/currencies");

            return array_keys($response->toArray());
        } catch (\Throwable) {
            return [];
        }
    }
}
```

```php
declare(strict_types=1);

namespace App\Infrastructure\FxRate;

use App\Domain\FxRate\FxRate;
use App\Domain\FxRate\FxRateId;
use App\Domain\FxRate\FxRateProviderInterface;
use App\Domain\FxRate\FxRateUnavailableException;
use App\Domain\Shared\ValueObject\CurrencyCode;
use Brick\Math\BigDecimal;

/**
 * Deterministic FX rates for tests — no HTTP calls.
 * All rates relative to USD.
 */
final class StubFxRateProvider implements FxRateProviderInterface
{
    /** @var array<string, float> Rates relative to USD */
    private array $rates = [
        'USD' => 1.0,
        'EUR' => 0.92,
        'GBP' => 0.79,
        'JPY' => 149.50,
        'CAD' => 1.36,
        'AUD' => 1.53,
        'CHF' => 0.88,
    ];

    public function withRate(string $from, string $to, float $rate): self
    {
        $clone = clone $this;
        $clone->rates["{$from}/{$to}"] = $rate;

        return $clone;
    }

    public function getRate(CurrencyCode $from, CurrencyCode $to): FxRate
    {
        $key = "{$from}/{$to}";

        if (isset($this->rates[$key])) {
            $rate = $this->rates[$key];
        } elseif (isset($this->rates[(string) $from]) && isset($this->rates[(string) $to])) {
            // Cross-rate via USD
            $rate = $this->rates[(string) $to] / $this->rates[(string) $from];
        } else {
            throw new FxRateUnavailableException($from, $to);
        }

        return new FxRate(
            id: FxRateId::generate(),
            baseCurrency: $from,
            targetCurrency: $to,
            rate: BigDecimal::of((string) round($rate, 10))->toScale(10),
            provider: 'stub',
            fetchedAt: new \DateTimeImmutable(),
        );
    }

    public function getSupportedCurrencies(): array
    {
        return array_keys($this->rates);
    }
}
```

```php
<?php

declare(strict_types=1);

namespace App\Infrastructure\FxRate;

use App\Domain\FxRate\FxRate;
use App\Domain\FxRate\FxRateId;
use App\Domain\FxRate\FxRateProviderInterface;
use App\Domain\FxRate\FxRateRepository;
use App\Domain\FxRate\FxRateUnavailableException;
use App\Domain\Shared\ValueObject\CurrencyCode;
use Brick\Math\BigDecimal;
use Psr\Cache\CacheItemPoolInterface;
use Psr\Log\LoggerInterface;

final class CachingFxRateService
{
    public function __construct(
        private readonly FxRateProviderInterface $provider,
        private readonly CacheItemPoolInterface $cache,
        private readonly FxRateRepository $fxRateRepository,
        private readonly LoggerInterface $logger,
        private readonly int $cacheTtl,
    ) {}

    public function getRate(CurrencyCode $from, CurrencyCode $to): FxRate
    {
        // 1. Same currency — trivial
        if ($from->value === $to->value) {
            return new FxRate(
                id: FxRateId::generate(),
                baseCurrency: $from,
                targetCurrency: $to,
                rate: BigDecimal::of('1')->toScale(10),
                provider: 'identity',
                fetchedAt: new \DateTimeImmutable(),
            );
        }

        // 2. Redis cache
        $cacheKey = "fx_rate_{$from}_{$to}";
        $item = $this->cache->getItem($cacheKey);

        if ($item->isHit()) {
            return $item->get();
        }

        // 3. Provider (Frankfurter)
        try {
            $rate = $this->provider->getRate($from, $to);
            $this->fxRateRepository->save($rate);

            $item->set($rate)->expiresAfter($this->cacheTtl);
            $this->cache->save($item);

            // Also cache stale version with longer TTL
            $staleItem = $this->cache->getItem("fx_rate_stale_{$from}_{$to}");
            $staleItem->set($rate)->expiresAfter($this->cacheTtl * 24);
            $this->cache->save($staleItem);

            return $rate;
        } catch (FxRateUnavailableException) {
            $this->logger->warning("FX provider unavailable for {$from}/{$to}, trying stale cache");
        }

        // 4. Stale Redis cache
        $staleItem = $this->cache->getItem("fx_rate_stale_{$from}_{$to}");
        if ($staleItem->isHit()) {
            $this->logger->warning("Using stale cached FX rate for {$from}/{$to}");

            return $staleItem->get();
        }

        // 5. DB fallback
        $fallback = $this->fxRateRepository->findLatest($from, $to);

        if ($fallback !== null) {
            $this->logger->warning("Using DB fallback FX rate for {$from}/{$to} (fetched at {$fallback->fetchedAt->format('Y-m-d H:i:s')})");

            return $fallback;
        }

        throw new FxRateUnavailableException($from, $to);
    }
}
```

```php
<?php

declare(strict_types=1);

namespace App\Controller\Api\V1;

use App\Domain\FxRate\FxRateUnavailableException;
use App\Domain\Shared\ValueObject\CurrencyCode;
use App\Infrastructure\FxRate\CachingFxRateService;
use Symfony\Component\HttpFoundation\JsonResponse;
use Symfony\Component\HttpKernel\Exception\UnprocessableEntityHttpException;
use Symfony\Component\Routing\Attribute\Route;

#[Route('/api/v1/fx-rates')]
final class FxRateController
{
    public function __construct(
        private readonly CachingFxRateService $fxRateService,
    ) {}

    #[Route('/{from}/{to}', methods: ['GET'])]
    public function pair(string $from, string $to): JsonResponse
    {
        try {
            $fromCurrency = new CurrencyCode(strtoupper($from));
            $toCurrency   = new CurrencyCode(strtoupper($to));
        } catch (\InvalidArgumentException $e) {
            throw new UnprocessableEntityHttpException($e->getMessage());
        }

        try {
            $rate = $this->fxRateService->getRate($fromCurrency, $toCurrency);
        } catch (FxRateUnavailableException $e) {
            throw new UnprocessableEntityHttpException($e->getMessage());
        }

        return new JsonResponse([
            'data' => [
                'base_currency'   => (string) $rate->baseCurrency,
                'target_currency' => (string) $rate->targetCurrency,
                'rate'            => (string) $rate->rate,
                'provider'        => $rate->provider,
                'fetched_at'      => $rate->fetchedAt->format(\DateTimeInterface::ATOM),
            ],
        ]);
    }
}
```
