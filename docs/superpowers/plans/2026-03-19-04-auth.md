# Fund Transfer API — Plan 4: Authentication

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement JWT authentication — register, login, refresh token, logout — with Argon2id password hashing, HaveIBeenPwned check, and rate limiting.

**Architecture:** LexikJWTAuthenticationBundle handles JWT signing (RS256). JWTRefreshTokenBundle manages refresh tokens. Registration and login are Application layer use cases. All auth endpoints are rate-limited via Symfony RateLimiter backed by Redis. Passwords hashed with Argon2id (OWASP minimum parameters).

**Tech Stack:** PHP 8.5, LexikJWTAuthenticationBundle, JWTRefreshTokenBundle, Symfony Security, Symfony RateLimiter, Redis, Pest

**Prerequisite:** Plans 1 and 2 complete.

```php
<?php

declare(strict_types=1);

namespace App\Infrastructure\Persistence;

use App\Domain\User\User;
use App\Domain\User\UserId;
use App\Domain\User\UserRepository;
use App\Domain\User\UserStatus;
use App\Domain\User\KycStatus;
use Doctrine\DBAL\Connection;

final class DoctrineUserRepository implements UserRepository
{
    public function __construct(private readonly Connection $connection) {}

    public function findById(UserId $id): ?User
    {
        $row = $this->connection->fetchAssociative(
            'SELECT * FROM users WHERE id = ?',
            [(string) $id]
        );

        return $row !== false ? $this->hydrate($row) : null;
    }

    public function findByEmail(string $email): ?User
    {
        $row = $this->connection->fetchAssociative(
            'SELECT * FROM users WHERE email = ?',
            [$email]
        );

        return $row !== false ? $this->hydrate($row) : null;
    }

    public function save(User $user): void
    {
        $this->connection->executeStatement(
            'INSERT INTO users
                (id, email, first_name, last_name, password_hash, phone_number,
                 date_of_birth, address, kyc_status, status,
                 email_verified_at, terms_accepted_at, created_at, updated_at)
             VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
             ON CONFLICT (id) DO UPDATE SET
                email = EXCLUDED.email,
                first_name = EXCLUDED.first_name,
                last_name = EXCLUDED.last_name,
                password_hash = EXCLUDED.password_hash,
                phone_number = EXCLUDED.phone_number,
                date_of_birth = EXCLUDED.date_of_birth,
                address = EXCLUDED.address,
                kyc_status = EXCLUDED.kyc_status,
                status = EXCLUDED.status,
                email_verified_at = EXCLUDED.email_verified_at,
                terms_accepted_at = EXCLUDED.terms_accepted_at,
                updated_at = EXCLUDED.updated_at',
            [
                (string) $user->id,
                $user->email,
                $user->firstName,
                $user->lastName,
                $user->passwordHash,
                $user->phoneNumber,
                $user->dateOfBirth?->format('Y-m-d'),
                $user->address !== null ? json_encode($user->address) : null,
                $user->kycStatus->value,
                $user->status->value,
                $user->emailVerifiedAt?->format('Y-m-d H:i:s'),
                $user->termsAcceptedAt?->format('Y-m-d H:i:s'),
                $user->createdAt->format('Y-m-d H:i:s'),
                $user->updatedAt->format('Y-m-d H:i:s'),
            ]
        );
    }

    public function nextIdentity(): UserId
    {
        return UserId::generate();
    }

    private function hydrate(array $row): User
    {
        // Use reflection to construct User from DB row
        // (User constructor is private — use ReflectionClass or a static factory)
        return User::fromRow($row); // see note below
    }
}
```

// src/Application/Auth/RegisterUserHandler.php
<?php

declare(strict_types=1);

namespace App\Application\Auth;

use App\Domain\User\User;
use App\Domain\User\UserRepository;
use App\Security\SecurityUser;
use Symfony\Component\PasswordHasher\Hasher\UserPasswordHasherInterface;

final class RegisterUserHandler
{
    public function __construct(
        private readonly UserRepository $userRepository,
        private readonly UserPasswordHasherInterface $passwordHasher,
    ) {}

    public function __invoke(RegisterUserCommand $command): User
    {
        if ($this->userRepository->findByEmail($command->email) !== null) {
            throw new \DomainException("Email '{$command->email}' is already registered.");
        }

        $id   = $this->userRepository->nextIdentity();
        $user = User::register(
            id: $id,
            email: $command->email,
            firstName: $command->firstName,
            lastName: $command->lastName,
            passwordHash: '', // temporary
            phoneNumber: $command->phoneNumber,
        );

        // Hash after creating user (hasher needs a UserInterface)
        $passwordHash = $this->passwordHasher->hashPassword(
            new SecurityUser($user),
            $command->password,
        );

        // Re-create with hash (or use a setter — see note)
        $user = User::register(
            id: $id,
            email: $command->email,
            firstName: $command->firstName,
            lastName: $command->lastName,
            passwordHash: $passwordHash,
            phoneNumber: $command->phoneNumber,
        );

        $this->userRepository->save($user);

        return $user;
    }
}

<?php

declare(strict_types=1);

namespace App\Controller\Api\V1;

use App\Application\Auth\RegisterUserCommand;
use App\Application\Auth\RegisterUserHandler;
use Lexik\Bundle\JWTAuthenticationBundle\Services\JWTTokenManagerInterface;
use App\Security\SecurityUser;
use Symfony\Component\HttpFoundation\JsonResponse;
use Symfony\Component\HttpFoundation\Request;
use Symfony\Component\HttpFoundation\Response;
use Symfony\Component\RateLimiter\RateLimiterFactory;
use Symfony\Component\Routing\Attribute\Route;
use Symfony\Component\Validator\Validator\ValidatorInterface;

#[Route('/api/v1/auth')]
final class AuthController
{
    public function __construct(
        private readonly RegisterUserHandler $registerUserHandler,
        private readonly JWTTokenManagerInterface $jwtManager,
        private readonly RateLimiterFactory $registerLimiter,
    ) {}

    #[Route('/register', methods: ['POST'])]
    public function register(Request $request): JsonResponse
    {
        $limiter = $this->registerLimiter->create($request->getClientIp());
        if (!$limiter->consume()->isAccepted()) {
            return new JsonResponse(
                ['type' => 'rate_limit_exceeded', 'title' => 'Too many requests'],
                Response::HTTP_TOO_MANY_REQUESTS
            );
        }

        $data = json_decode($request->getContent(), true) ?? [];

        try {
            $user = ($this->registerUserHandler)(new RegisterUserCommand(
                email: $data['email'] ?? '',
                password: $data['password'] ?? '',
                firstName: $data['first_name'] ?? '',
                lastName: $data['last_name'] ?? '',
            ));
        } catch (\DomainException $e) {
            return new JsonResponse(
                ['type' => 'conflict', 'title' => $e->getMessage()],
                Response::HTTP_CONFLICT
            );
        }

        $securityUser = new SecurityUser($user);
        $token = $this->jwtManager->create($securityUser);

        return new JsonResponse(
            ['data' => ['token' => $token, 'refresh_token' => null]], // refresh_token handled by bundle
            Response::HTTP_CREATED
        );
    }
}
