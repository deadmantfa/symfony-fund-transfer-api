```php
<?php

declare(strict_types=1);

namespace App\Infrastructure\Persistence;

use App\Domain\Account\Account;
use App\Domain\Account\AccountId;
use App\Domain\Account\AccountRepository;
use App\Domain\Account\AccountStatus;
use App\Domain\Shared\ValueObject\CurrencyCode;
use App\Domain\Shared\ValueObject\Money;
use App\Domain\User\UserId;
use Brick\Math\BigDecimal;
use Doctrine\DBAL\Connection;
use Doctrine\DBAL\LockMode;

final class DoctrineAccountRepository implements AccountRepository
{
    public function __construct(private readonly Connection $connection) {}

    public function findById(AccountId $id): ?Account
    {
        $row = $this->connection->fetchAssociative(
            'SELECT * FROM accounts WHERE id = ?',
            [(string) $id]
        );

        return $row !== false ? $this->hydrate($row) : null;
    }

    public function findByIdForUpdate(AccountId $id): ?Account
    {
        $row = $this->connection->fetchAssociative(
            'SELECT * FROM accounts WHERE id = ? FOR UPDATE',
            [(string) $id]
        );

        return $row !== false ? $this->hydrate($row) : null;
    }

    public function findByUserId(UserId $userId): array
    {
        $rows = $this->connection->fetchAllAssociative(
            'SELECT * FROM accounts WHERE user_id = ? ORDER BY created_at ASC',
            [(string) $userId]
        );

        return array_map($this->hydrate(...), $rows);
    }

    public function save(Account $account): void
    {
        $this->connection->executeStatement(
            'INSERT INTO accounts (id, user_id, currency, balance, name, status, created_at, updated_at)
             VALUES (?, ?, ?, ?, ?, ?, ?, ?)
             ON CONFLICT (id) DO UPDATE SET
                balance = EXCLUDED.balance,
                name = EXCLUDED.name,
                status = EXCLUDED.status,
                updated_at = EXCLUDED.updated_at',
            [
                (string) $account->id,
                (string) $account->userId,
                (string) $account->currency,
                (string) $account->balance->amount,
                $account->name,
                $account->status->value,
                $account->createdAt->format('Y-m-d H:i:s'),
                $account->updatedAt->format('Y-m-d H:i:s'),
            ]
        );
    }

    public function nextIdentity(): AccountId
    {
        return AccountId::generate();
    }

    public function existsForUserAndCurrency(UserId $userId, string $currency): bool
    {
        $count = $this->connection->fetchOne(
            'SELECT COUNT(*) FROM accounts WHERE user_id = ? AND currency = ? AND status != ?',
            [(string) $userId, $currency, AccountStatus::Closed->value]
        );

        return (int) $count > 0;
    }

    private function hydrate(array $row): Account
    {
        return Account::fromRow($row);
    }
}
```

    public function append(LedgerEntry $entry): void
    {
        $this->connection->executeStatement(
            'INSERT INTO ledger_entries
                (id, transfer_id, account_id, entry_type, amount, currency,
                 balance_before, balance_after, created_at)
             VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)',
            [
                (string) $entry->id,
                (string) $entry->transferId,
                (string) $entry->accountId,
                $entry->entryType->value,
                (string) $entry->amount->amount,
                (string) $entry->amount->currency,
                (string) $entry->balanceBefore->amount,
                (string) $entry->balanceAfter->amount,
                $entry->createdAt->format('Y-m-d H:i:s'),
            ]
        );
    }

    public function findByAccountId(AccountId $accountId, ?string $cursor, int $limit): array
    {
        $sql    = 'SELECT * FROM ledger_entries WHERE account_id = ?';
        $params = [(string) $accountId];

        if ($cursor !== null) {
            $sql      .= ' AND created_at < ?';
            $params[] = $cursor;
        }

        $sql .= ' ORDER BY created_at DESC LIMIT ?';
        $params[] = $limit + 1; // fetch one extra to determine if there's a next page

        $rows = $this->connection->fetchAllAssociative($sql, $params);

        return array_map(fn (array $row) => new LedgerEntry(
            // ... hydrate
        ), $rows);
    }

    public function nextIdentity(): LedgerEntryId
    {
        return LedgerEntryId::generate();
    }
}

<?php

declare(strict_types=1);

namespace App\Security\Voter;

use App\Domain\Account\Account;
use App\Domain\User\UserId;
use App\Security\SecurityUser;
use Symfony\Component\Security\Core\Authentication\Token\TokenInterface;
use Symfony\Component\Security\Core\Authorization\Voter\Voter;

final class AccountVoter extends Voter
{
    public const VIEW   = 'account.view';
    public const EDIT   = 'account.edit';
    public const DELETE = 'account.delete';

    protected function supports(string $attribute, mixed $subject): bool
    {
        return in_array($attribute, [self::VIEW, self::EDIT, self::DELETE], true)
            && $subject instanceof Account;
    }

    protected function voteOnAttribute(string $attribute, mixed $subject, TokenInterface $token): bool
    {
        $securityUser = $token->getUser();

        if (!$securityUser instanceof SecurityUser) {
            return false;
        }

        $userId = new UserId((string) $securityUser->user->id);

        return $subject->isOwnedBy($userId);
    }
}

<?php

declare(strict_types=1);

namespace App\Application\Account;

use App\Domain\Account\Account;
use App\Domain\Account\AccountRepository;
use App\Domain\Shared\ValueObject\CurrencyCode;
use App\Domain\User\UserId;

final class CreateAccountHandler
{
    public function __construct(private readonly AccountRepository $accountRepository) {}

    public function __invoke(CreateAccountCommand $command): Account
    {
        $userId   = new UserId($command->userId);
        $currency = new CurrencyCode(strtoupper($command->currency));

        if ($this->accountRepository->existsForUserAndCurrency($userId, $command->currency)) {
            throw new \DomainException("User already has a {$currency} account.");
        }

        $account = Account::open(
            id: $this->accountRepository->nextIdentity(),
            userId: $userId,
            currency: $currency,
            name: $command->name,
        );

        $this->accountRepository->save($account);

        return $account;
    }
}

<?php

declare(strict_types=1);

namespace App\Controller\Api\V1;

use App\Application\Account\CloseAccountCommand;
use App\Application\Account\CloseAccountHandler;
use App\Application\Account\CreateAccountCommand;
use App\Application\Account\CreateAccountHandler;
use App\Domain\Account\AccountId;
use App\Domain\Account\AccountRepository;
use App\Security\SecurityUser;
use App\Security\Voter\AccountVoter;
use Symfony\Bundle\FrameworkBundle\Controller\AbstractController;
use Symfony\Component\HttpFoundation\JsonResponse;
use Symfony\Component\HttpFoundation\Request;
use Symfony\Component\HttpFoundation\Response;
use Symfony\Component\Routing\Attribute\Route;

#[Route('/api/v1/accounts')]
final class AccountController extends AbstractController
{
    public function __construct(
        private readonly CreateAccountHandler $createAccountHandler,
        private readonly CloseAccountHandler $closeAccountHandler,
        private readonly AccountRepository $accountRepository,
    ) {}

    #[Route('', methods: ['GET'])]
    public function list(): JsonResponse
    {
        /** @var SecurityUser $securityUser */
        $securityUser = $this->getUser();
        $accounts = $this->accountRepository->findByUserId($securityUser->user->id);

        return new JsonResponse([
            'data' => array_map($this->serializeAccount(...), $accounts),
        ]);
    }

    #[Route('', methods: ['POST'])]
    public function create(Request $request): JsonResponse
    {
        /** @var SecurityUser $securityUser */
        $securityUser = $this->getUser();
        $data = json_decode($request->getContent(), true) ?? [];

        try {
            $account = ($this->createAccountHandler)(new CreateAccountCommand(
                userId: (string) $securityUser->user->id,
                currency: $data['currency'] ?? '',
                name: $data['name'] ?? '',
            ));
        } catch (\DomainException $e) {
            return new JsonResponse(
                ['type' => 'conflict', 'title' => $e->getMessage()],
                Response::HTTP_CONFLICT
            );
        }

        return new JsonResponse(['data' => $this->serializeAccount($account)], Response::HTTP_CREATED);
    }

    #[Route('/{id}', methods: ['GET'])]
    public function get(string $id): JsonResponse
    {
        $account = $this->accountRepository->findById(new AccountId($id));

        if ($account === null) {
            return new JsonResponse(['type' => 'not_found', 'title' => 'Account not found'], 404);
        }

        $this->denyAccessUnlessGranted(AccountVoter::VIEW, $account);

        return new JsonResponse(['data' => $this->serializeAccount($account)]);
    }

    #[Route('/{id}', methods: ['DELETE'])]
    public function close(string $id): JsonResponse
    {
        $account = $this->accountRepository->findById(new AccountId($id));

        if ($account === null) {
            return new JsonResponse(['type' => 'not_found', 'title' => 'Account not found'], 404);
        }

        $this->denyAccessUnlessGranted(AccountVoter::DELETE, $account);

        try {
            ($this->closeAccountHandler)(new CloseAccountCommand($id));
        } catch (\DomainException $e) {
            return new JsonResponse(
                ['type' => 'conflict', 'title' => $e->getMessage()],
                Response::HTTP_CONFLICT
            );
        }

        return new JsonResponse(null, Response::HTTP_NO_CONTENT);
    }

    private function serializeAccount(\App\Domain\Account\Account $account): array
    {
        return [
            'id'         => (string) $account->id,
            'currency'   => (string) $account->currency,
            'name'       => $account->name,
            'balance'    => (string) $account->balance->amount,
            'status'     => $account->status->value,
            'created_at' => $account->createdAt->format(\DateTimeInterface::ATOM),
        ];
    }
}
