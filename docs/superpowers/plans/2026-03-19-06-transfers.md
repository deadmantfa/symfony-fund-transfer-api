# Fund Transfer API — Plan 6: Transfer System

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement the async transfer system — idempotency, initiation (202 Accepted), Symfony Messenger async processing, pessimistic locking, FX conversion, double-entry ledger, and concurrency tests.

**Architecture:** `InitiateTransferHandler` validates, creates a pending transfer, and dispatches `ProcessTransferCommand` to Redis queue. `ProcessTransferHandler` runs in the worker container: acquires pessimistic locks on both accounts in ascending UUID order (deadlock prevention), fetches FX rate, debits source, credits destination, writes two immutable ledger entries, and sets transfer status to completed. Idempotency is enforced via Redis (primary) and DB deduplication (secondary).

**Tech Stack:** PHP 8.5, Symfony Messenger (Redis transport), Doctrine DBAL, Redis, Pest (including concurrency tests)

**Prerequisite:** Plans 1–5 complete.

```php
<?php

declare(strict_types=1);

namespace App\Application\Shared;

use Psr\Cache\CacheItemPoolInterface;

final class IdempotencyService
{
    public function __construct(
        private readonly CacheItemPoolInterface $cache,
        private readonly int $ttl,
    ) {}

    public function exists(string $key): bool
    {
        return $this->cache->getItem("idempotency_{$key}")->isHit();
    }

    public function get(string $key): ?array
    {
        $item = $this->cache->getItem("idempotency_{$key}");

        return $item->isHit() ? $item->get() : null;
    }

    public function store(string $key, array $value): void
    {
        $item = $this->cache->getItem("idempotency_{$key}");
        $item->set($value)->expiresAfter($this->ttl);
        $this->cache->save($item);
    }
}
```

```php
<?php

declare(strict_types=1);

namespace App\Infrastructure\Persistence;

use App\Domain\Account\AccountId;
use App\Domain\Transfer\Transfer;
use App\Domain\Transfer\TransferId;
use App\Domain\Transfer\TransferRepository;
use App\Domain\Transfer\TransferStatus;
use Doctrine\DBAL\Connection;

final class DoctrineTransferRepository implements TransferRepository
{
    public function __construct(private readonly Connection $connection) {}

    public function findById(TransferId $id): ?Transfer
    {
        $row = $this->connection->fetchAssociative(
            'SELECT * FROM transfers WHERE id = ?',
            [(string) $id]
        );

        return $row !== false ? Transfer::fromRow($row) : null;
    }

    public function findByIdForUpdate(TransferId $id): ?Transfer
    {
        $row = $this->connection->fetchAssociative(
            'SELECT * FROM transfers WHERE id = ? FOR UPDATE',
            [(string) $id]
        );

        return $row !== false ? Transfer::fromRow($row) : null;
    }

    public function findByIdempotencyKey(string $key): ?Transfer
    {
        $row = $this->connection->fetchAssociative(
            'SELECT * FROM transfers WHERE idempotency_key = ?',
            [$key]
        );

        return $row !== false ? Transfer::fromRow($row) : null;
    }

    public function findByAccountId(AccountId $accountId, ?string $cursor, int $limit): array
    {
        $sql    = 'SELECT * FROM transfers WHERE (from_account_id = ? OR to_account_id = ?)';
        $params = [(string) $accountId, (string) $accountId];

        if ($cursor !== null) {
            $sql      .= ' AND created_at < ?';
            $params[] = $cursor;
        }

        $sql .= ' ORDER BY created_at DESC LIMIT ?';
        $params[] = $limit + 1;

        $rows = $this->connection->fetchAllAssociative($sql, $params);

        return array_map(Transfer::fromRow(...), $rows);
    }

    public function save(Transfer $transfer): void
    {
        $this->connection->executeStatement(
            'INSERT INTO transfers
                (id, from_account_id, to_account_id, amount, currency,
                 from_amount, from_currency, to_amount, to_currency,
                 fx_rate, fx_rate_provider, status, idempotency_key,
                 failure_reason, initiated_by_user_id, ip_address, user_agent,
                 created_at, updated_at)
             VALUES (?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?)
             ON CONFLICT (id) DO UPDATE SET
                from_amount = EXCLUDED.from_amount,
                from_currency = EXCLUDED.from_currency,
                to_amount = EXCLUDED.to_amount,
                to_currency = EXCLUDED.to_currency,
                fx_rate = EXCLUDED.fx_rate,
                fx_rate_provider = EXCLUDED.fx_rate_provider,
                status = EXCLUDED.status,
                failure_reason = EXCLUDED.failure_reason,
                updated_at = EXCLUDED.updated_at',
            [
                (string) $transfer->id,
                (string) $transfer->fromAccountId,
                (string) $transfer->toAccountId,
                (string) $transfer->amount->amount,
                (string) $transfer->amount->currency,
                $transfer->fromAmount !== null ? (string) $transfer->fromAmount->amount : null,
                $transfer->fromAmount !== null ? (string) $transfer->fromAmount->currency : null,
                $transfer->toAmount !== null ? (string) $transfer->toAmount->amount : null,
                $transfer->toAmount !== null ? (string) $transfer->toAmount->currency : null,
                $transfer->fxRate,
                $transfer->fxRateProvider,
                $transfer->status->value,
                $transfer->idempotencyKey,
                $transfer->failureReason,
                (string) $transfer->initiatedByUserId,
                $transfer->ipAddress,
                $transfer->userAgent,
                $transfer->createdAt->format('Y-m-d H:i:s'),
                $transfer->updatedAt->format('Y-m-d H:i:s'),
            ]
        );
    }

    public function nextIdentity(): TransferId
    {
        return TransferId::generate();
    }

    public function existsDuplicate(
        AccountId $fromAccountId,
        AccountId $toAccountId,
        string $amount,
        string $currency,
        int $withinSeconds,
    ): bool {
        $count = $this->connection->fetchOne(
            'SELECT COUNT(*) FROM transfers
             WHERE from_account_id = ?
               AND to_account_id = ?
               AND amount = ?
               AND currency = ?
               AND status NOT IN (?, ?)
               AND created_at > NOW() - INTERVAL \'' . $withinSeconds . ' seconds\'',
            [
                (string) $fromAccountId,
                (string) $toAccountId,
                $amount,
                $currency,
                TransferStatus::Failed->value,
                TransferStatus::Completed->value,
            ]
        );

        return (int) $count > 0;
    }
}
```

```php
// src/Application/Transfer/InitiateTransferHandler.php
<?php

declare(strict_types=1);

namespace App\Application\Transfer;

use App\Application\Shared\IdempotencyService;
use App\Domain\Account\AccountId;
use App\Domain\Account\AccountRepository;
use App\Domain\Shared\ValueObject\CurrencyCode;
use App\Domain\Shared\ValueObject\Money;
use App\Domain\Transfer\Transfer;
use App\Domain\Transfer\TransferRepository;
use App\Domain\User\UserId;
use App\Domain\User\UserRepository;
use Symfony\Component\Messenger\MessageBusInterface;

final class InitiateTransferHandler
{
    public function __construct(
        private readonly UserRepository $userRepository,
        private readonly AccountRepository $accountRepository,
        private readonly TransferRepository $transferRepository,
        private readonly IdempotencyService $idempotencyService,
        private readonly MessageBusInterface $messageBus,
    ) {}

    public function __invoke(InitiateTransferCommand $command): Transfer
    {
        // 1. Check idempotency key (Redis)
        if ($this->idempotencyService->exists($command->idempotencyKey)) {
            throw new DuplicateIdempotencyKeyException(
                $this->idempotencyService->get($command->idempotencyKey) ?? []
            );
        }

        // 2. Load and validate user
        $userId = new UserId($command->userId);
        $user   = $this->userRepository->findById($userId);

        if ($user === null) {
            throw new \DomainException('User not found.');
        }

        if (!$user->canInitiateTransfer()) {
            throw new \DomainException('User cannot initiate transfers. KYC verification required.');
        }

        // 3. Load and validate accounts
        $fromAccountId = new AccountId($command->fromAccountId);
        $toAccountId   = new AccountId($command->toAccountId);

        $fromAccount = $this->accountRepository->findById($fromAccountId);
        $toAccount   = $this->accountRepository->findById($toAccountId);

        if ($fromAccount === null || $toAccount === null) {
            throw new \DomainException('One or both accounts not found.');
        }

        if (!$fromAccount->isOwnedBy($userId)) {
            throw new \DomainException('You do not own the source account.');
        }

        if (!$fromAccount->status->allowsTransactions()) {
            throw new \DomainException("Source account is {$fromAccount->status->value}.");
        }

        if (!$toAccount->status->allowsTransactions()) {
            throw new \DomainException("Destination account is {$toAccount->status->value}.");
        }

        // 4. Server-side dedup (same params within 60s)
        if ($this->transferRepository->existsDuplicate(
            $fromAccountId, $toAccountId, $command->amount, $command->currency, 60
        )) {
            throw new \DomainException('Duplicate transfer detected. Please wait before retrying.');
        }

        // 5. Create pending transfer
        $transfer = Transfer::initiate(
            id: $this->transferRepository->nextIdentity(),
            fromAccountId: $fromAccountId,
            toAccountId: $toAccountId,
            amount: new Money($command->amount, new CurrencyCode($command->currency)),
            idempotencyKey: $command->idempotencyKey,
            initiatedByUserId: $userId,
            ipAddress: $command->ipAddress,
            userAgent: $command->userAgent,
        );

        $this->transferRepository->save($transfer);

        // 6. Cache idempotency key
        $this->idempotencyService->store($command->idempotencyKey, [
            'transfer_id' => (string) $transfer->id,
            'status'      => $transfer->status->value,
        ]);

        // 7. Dispatch to worker queue
        $this->messageBus->dispatch(new ProcessTransferCommand((string) $transfer->id));

        return $transfer;
    }
}
```

```php
<?php

declare(strict_types=1);

namespace App\Application\Transfer;

use App\Domain\Account\AccountRepository;
use App\Domain\AuditLog\AuditLog;
use App\Domain\AuditLog\AuditLogRepository;
use App\Domain\FxRate\FxRateUnavailableException;
use App\Domain\LedgerEntry\EntryType;
use App\Domain\LedgerEntry\LedgerEntry;
use App\Domain\LedgerEntry\LedgerEntryRepository;
use App\Domain\Transfer\Transfer;
use App\Domain\Transfer\TransferId;
use App\Domain\Transfer\TransferRepository;
use App\Infrastructure\FxRate\CachingFxRateService;
use Doctrine\DBAL\Connection;

final class ProcessTransferHandler
{
    public function __construct(
        private readonly Connection $connection,
        private readonly TransferRepository $transferRepository,
        private readonly AccountRepository $accountRepository,
        private readonly LedgerEntryRepository $ledgerEntryRepository,
        private readonly AuditLogRepository $auditLogRepository,
        private readonly CachingFxRateService $fxRateService,
    ) {}

    public function __invoke(ProcessTransferCommand $command): void
    {
        $this->connection->beginTransaction();

        try {
            // 1. Lock transfer row
            $transfer = $this->transferRepository->findByIdForUpdate(
                new TransferId($command->transferId)
            );

            if ($transfer === null) {
                throw new \RuntimeException("Transfer {$command->transferId} not found.");
            }

            $transfer->markProcessing();
            $this->transferRepository->save($transfer);

            // 2. Lock accounts in ascending UUID order (deadlock prevention)
            $ids = [(string) $transfer->fromAccountId, (string) $transfer->toAccountId];
            sort($ids);

            $first  = $this->accountRepository->findByIdForUpdate(
                $ids[0] === (string) $transfer->fromAccountId
                    ? $transfer->fromAccountId
                    : $transfer->toAccountId
            );
            $second = $this->accountRepository->findByIdForUpdate(
                $ids[0] === (string) $transfer->fromAccountId
                    ? $transfer->toAccountId
                    : $transfer->fromAccountId
            );

            $fromAccount = $first->id->equals($transfer->fromAccountId) ? $first : $second;
            $toAccount   = $first->id->equals($transfer->toAccountId) ? $first : $second;

            if ($fromAccount === null || $toAccount === null) {
                throw new \DomainException('Account not found during processing.');
            }

            // 3. Resolve FX rate
            $fxRate = $this->fxRateService->getRate(
                $fromAccount->currency,
                $toAccount->currency,
            );

            // 4. Calculate amounts
            $fromAmount = $transfer->amount->multiply($fxRate->rate->isEqualTo('1')
                ? '1'
                : (string) ($fromAccount->currency->value === $transfer->amount->currency->value
                    ? '1'
                    : $fxRate->rate));

            // Transfer amount is in transfer currency; convert to from/to currencies
            $transferAmountInFromCurrency = $transfer->amount->currency->value === $fromAccount->currency->value
                ? $transfer->amount
                : $transfer->amount->multiply((string) $this->fxRateService->getRate($transfer->amount->currency, $fromAccount->currency)->rate);

            $toAmount = $transferAmountInFromCurrency->multiply((string) $fxRate->rate);

            // 5. Debit source account
            $balanceBeforeDebit = $fromAccount->balance;
            $fromAccount->debit($transferAmountInFromCurrency); // throws DomainException on insufficient funds
            $this->accountRepository->save($fromAccount);

            // 6. Credit destination account
            $balanceBeforeCredit = $toAccount->balance;
            $toAccount->credit($toAmount);
            $this->accountRepository->save($toAccount);

            // 7. Write double-entry ledger (immutable)
            $this->ledgerEntryRepository->append(LedgerEntry::record(
                id: $this->ledgerEntryRepository->nextIdentity(),
                transferId: $transfer->id,
                accountId: $fromAccount->id,
                entryType: EntryType::Debit,
                amount: $transferAmountInFromCurrency,
                balanceBefore: $balanceBeforeDebit,
                balanceAfter: $fromAccount->balance,
            ));

            $this->ledgerEntryRepository->append(LedgerEntry::record(
                id: $this->ledgerEntryRepository->nextIdentity(),
                transferId: $transfer->id,
                accountId: $toAccount->id,
                entryType: EntryType::Credit,
                amount: $toAmount,
                balanceBefore: $balanceBeforeCredit,
                balanceAfter: $toAccount->balance,
            ));

            // 8. Complete transfer
            $transfer->complete(
                fromAmount: $transferAmountInFromCurrency,
                toAmount: $toAmount,
                fxRate: (string) $fxRate->rate,
                fxRateProvider: $fxRate->provider,
            );
            $this->transferRepository->save($transfer);

            // 9. Audit log
            $this->auditLogRepository->append(AuditLog::record(
                id: $this->auditLogRepository->nextIdentity(),
                entityType: 'Transfer',
                entityId: (string) $transfer->id,
                action: 'completed',
                actorId: $transfer->initiatedByUserId,
                newValue: ['status' => 'completed', 'fx_rate' => (string) $fxRate->rate],
                ipAddress: $transfer->ipAddress,
                userAgent: $transfer->userAgent,
            ));

            $this->connection->commit();
        } catch (\DomainException $e) {
            $this->connection->rollBack();

            // Mark as failed (non-retryable)
            $transfer?->fail(match (true) {
                str_contains($e->getMessage(), 'Insufficient') => 'insufficient_funds',
                str_contains($e->getMessage(), 'frozen')       => 'account_frozen',
                str_contains($e->getMessage(), 'closed')       => 'account_closed',
                default                                         => 'processing_error',
            });

            if (isset($transfer)) {
                $this->transferRepository->save($transfer);
                $this->auditLogRepository->append(AuditLog::record(
                    id: $this->auditLogRepository->nextIdentity(),
                    entityType: 'Transfer',
                    entityId: (string) $transfer->id,
                    action: 'failed',
                    newValue: ['reason' => $transfer->failureReason, 'error' => $e->getMessage()],
                ));
            }
        } catch (FxRateUnavailableException $e) {
            // Retryable — re-throw so Messenger retries
            $this->connection->rollBack();
            throw $e;
        } catch (\Throwable $e) {
            $this->connection->rollBack();
            throw $e;
        }
    }
}
```

```php
use Symfony\Component\Security\Core\Authentication\Token\TokenInterface;
use Symfony\Component\Security\Core\Authorization\Voter\Voter;

final class TransferVoter extends Voter
{
    public const VIEW   = 'transfer.view';
    public const CANCEL = 'transfer.cancel';

    protected function supports(string $attribute, mixed $subject): bool
    {
        return in_array($attribute, [self::VIEW, self::CANCEL], true)
            && $subject instanceof Transfer;
    }

    protected function voteOnAttribute(string $attribute, mixed $subject, TokenInterface $token): bool
    {
        $securityUser = $token->getUser();
        if (!$securityUser instanceof SecurityUser) {
            return false;
        }

        $userId = new UserId((string) $securityUser->user->id);

        // User can view/cancel if they own the source account
        return $subject->initiatedByUserId->equals($userId);
    }
}
```

```php
<?php

declare(strict_types=1);

namespace App\Controller\Api\V1;

use App\Application\Transfer\DuplicateIdempotencyKeyException;
use App\Application\Transfer\InitiateTransferCommand;
use App\Application\Transfer\InitiateTransferHandler;
use App\Domain\Transfer\TransferId;
use App\Domain\Transfer\TransferRepository;
use App\Security\SecurityUser;
use App\Security\Voter\TransferVoter;
use Symfony\Bundle\FrameworkBundle\Controller\AbstractController;
use Symfony\Component\HttpFoundation\JsonResponse;
use Symfony\Component\HttpFoundation\Request;
use Symfony\Component\HttpFoundation\Response;
use Symfony\Component\Routing\Attribute\Route;

#[Route('/api/v1/transfers')]
final class TransferController extends AbstractController
{
    public function __construct(
        private readonly InitiateTransferHandler $initiateTransferHandler,
        private readonly TransferRepository $transferRepository,
    ) {}

    #[Route('', methods: ['POST'])]
    public function initiate(Request $request): JsonResponse
    {
        $idempotencyKey = $request->headers->get('Idempotency-Key');

        if (empty($idempotencyKey)) {
            return new JsonResponse(
                ['type' => 'validation_error', 'title' => 'Idempotency-Key header is required'],
                Response::HTTP_UNPROCESSABLE_ENTITY
            );
        }

        $data = json_decode($request->getContent(), true) ?? [];

        /** @var SecurityUser $securityUser */
        $securityUser = $this->getUser();

        try {
            $transfer = ($this->initiateTransferHandler)(new InitiateTransferCommand(
                userId: (string) $securityUser->user->id,
                fromAccountId: $data['from_account_id'] ?? '',
                toAccountId: $data['to_account_id'] ?? '',
                amount: $data['amount'] ?? '',
                currency: $data['currency'] ?? '',
                idempotencyKey: $idempotencyKey,
                ipAddress: $request->getClientIp() ?? '',
                userAgent: $request->headers->get('User-Agent', ''),
                note: $data['note'] ?? null,
            ));
        } catch (DuplicateIdempotencyKeyException $e) {
            return new JsonResponse(['data' => $e->cachedResponse], Response::HTTP_OK);
        } catch (\DomainException $e) {
            return new JsonResponse(
                ['type' => 'conflict', 'title' => $e->getMessage()],
                Response::HTTP_CONFLICT
            );
        }

        return new JsonResponse(
            ['data' => $this->serializeTransfer($transfer)],
            Response::HTTP_ACCEPTED,
            ['Location' => "/api/v1/transfers/{$transfer->id}"]
        );
    }

    #[Route('/{id}', methods: ['GET'])]
    public function get(string $id): JsonResponse
    {
        $transfer = $this->transferRepository->findById(new TransferId($id));

        if ($transfer === null) {
            return new JsonResponse(['type' => 'not_found', 'title' => 'Transfer not found'], 404);
        }

        $this->denyAccessUnlessGranted(TransferVoter::VIEW, $transfer);

        return new JsonResponse(['data' => $this->serializeTransfer($transfer)]);
    }

    #[Route('/{id}', methods: ['DELETE'])]
    public function cancel(string $id): JsonResponse
    {
        $transfer = $this->transferRepository->findById(new TransferId($id));

        if ($transfer === null) {
            return new JsonResponse(['type' => 'not_found', 'title' => 'Transfer not found'], 404);
        }

        $this->denyAccessUnlessGranted(TransferVoter::CANCEL, $transfer);

        try {
            $transfer->cancel();
            $this->transferRepository->save($transfer);
        } catch (\DomainException $e) {
            return new JsonResponse(
                ['type' => 'conflict', 'title' => $e->getMessage()],
                Response::HTTP_CONFLICT
            );
        }

        return new JsonResponse(['data' => $this->serializeTransfer($transfer)]);
    }

    private function serializeTransfer(\App\Domain\Transfer\Transfer $transfer): array
    {
        return [
            'id'               => (string) $transfer->id,
            'from_account_id'  => (string) $transfer->fromAccountId,
            'to_account_id'    => (string) $transfer->toAccountId,
            'amount'           => (string) $transfer->amount->amount,
            'currency'         => (string) $transfer->amount->currency,
            'from_amount'      => $transfer->fromAmount !== null ? (string) $transfer->fromAmount->amount : null,
            'from_currency'    => $transfer->fromAmount !== null ? (string) $transfer->fromAmount->currency : null,
            'to_amount'        => $transfer->toAmount !== null ? (string) $transfer->toAmount->amount : null,
            'to_currency'      => $transfer->toAmount !== null ? (string) $transfer->toAmount->currency : null,
            'fx_rate'          => $transfer->fxRate,
            'status'           => $transfer->status->value,
            'failure_reason'   => $transfer->failureReason,
            'created_at'       => $transfer->createdAt->format(\DateTimeInterface::ATOM),
            'updated_at'       => $transfer->updatedAt->format(\DateTimeInterface::ATOM),
        ];
    }
}
```
