```php
<?php

declare(strict_types=1);

namespace App\EventListener;

use Symfony\Component\EventDispatcher\Attribute\AsEventListener;
use Symfony\Component\HttpFoundation\JsonResponse;
use Symfony\Component\HttpKernel\Event\ExceptionEvent;
use Symfony\Component\HttpKernel\Exception\HttpExceptionInterface;
use Symfony\Component\HttpKernel\KernelEvents;

/**
 * Normalizes all exceptions to RFC 9457 Problem Details format.
 * Content-Type: application/problem+json
 * Never exposes stack traces.
 */
#[AsEventListener(event: KernelEvents::EXCEPTION, priority: -1)]
final class ExceptionListener
{
    public function __invoke(ExceptionEvent $event): void
    {
        $exception = $event->getThrowable();
        $request   = $event->getRequest();

        // Only handle API routes
        if (!str_starts_with($request->getPathInfo(), '/api') && $request->getPathInfo() !== '/health') {
            return;
        }

        $statusCode = $exception instanceof HttpExceptionInterface
            ? $exception->getStatusCode()
            : 500;

        $problem = [
            'type'   => $this->mapType($statusCode),
            'title'  => $this->mapTitle($statusCode, $exception),
            'status' => $statusCode,
        ];

        // Add detail for 4xx errors (safe to expose)
        if ($statusCode >= 400 && $statusCode < 500 && $exception->getMessage()) {
            $problem['detail'] = $exception->getMessage();
        }

        // Never expose internals in 5xx
        if ($statusCode >= 500) {
            $problem['detail'] = 'An internal error occurred.';
        }

        $response = new JsonResponse($problem, $statusCode);
        $response->headers->set('Content-Type', 'application/problem+json');

        $event->setResponse($response);
    }

    private function mapType(int $status): string
    {
        return match ($status) {
            400 => 'https://tools.ietf.org/html/rfc9110#section-15.5.1',
            401 => 'https://tools.ietf.org/html/rfc9110#section-15.5.2',
            403 => 'https://tools.ietf.org/html/rfc9110#section-15.5.4',
            404 => 'https://tools.ietf.org/html/rfc9110#section-15.5.5',
            409 => 'https://tools.ietf.org/html/rfc9110#section-15.5.10',
            422 => 'https://tools.ietf.org/html/rfc9110#section-15.5.21',
            429 => 'https://tools.ietf.org/html/rfc6585#section-4',
            default => 'https://tools.ietf.org/html/rfc9110#section-15.6.1',
        };
    }

    private function mapTitle(int $status, \Throwable $e): string
    {
        return match ($status) {
            400 => 'Bad Request',
            401 => 'Unauthorized',
            403 => 'Forbidden',
            404 => 'Not Found',
            409 => 'Conflict',
            422 => 'Unprocessable Content',
            429 => 'Too Many Requests',
            500 => 'Internal Server Error',
            default => $e->getMessage() ?: 'Error',
        };
    }
}
```

```php
<?php

declare(strict_types=1);

namespace App\Infrastructure\Encryption;

/**
 * Libsodium-based symmetric encryption for PII fields.
 * Uses XSalsa20-Poly1305 via sodium_crypto_secretbox.
 * Each encryption generates a random 24-byte nonce.
 */
final class EncryptionService
{
    private readonly string $key;

    public function __construct(string $keyHex)
    {
        if (strlen($keyHex) !== SODIUM_CRYPTO_SECRETBOX_KEYBYTES * 2) {
            throw new \InvalidArgumentException(
                'Encryption key must be a ' . (SODIUM_CRYPTO_SECRETBOX_KEYBYTES * 2) . '-char hex string.'
            );
        }

        $this->key = sodium_hex2bin($keyHex);
    }

    public function encrypt(string $plaintext): string
    {
        $nonce      = random_bytes(SODIUM_CRYPTO_SECRETBOX_NONCEBYTES);
        $ciphertext = sodium_crypto_secretbox($plaintext, $nonce, $this->key);

        return sodium_bin2base64($nonce . $ciphertext, SODIUM_BASE64_VARIANT_URLSAFE);
    }

    public function decrypt(string $encoded): ?string
    {
        try {
            $decoded   = sodium_base642bin($encoded, SODIUM_BASE64_VARIANT_URLSAFE);
            $nonce     = substr($decoded, 0, SODIUM_CRYPTO_SECRETBOX_NONCEBYTES);
            $ciphertext = substr($decoded, SODIUM_CRYPTO_SECRETBOX_NONCEBYTES);
            $plaintext = sodium_crypto_secretbox_open($ciphertext, $nonce, $this->key);

            return $plaintext === false ? null : $plaintext;
        } catch (\Throwable) {
            return null;
        }
    }
}
```

## Task 6: Seed command

**Files:**
- Create: `src/Command/SeedCommand.php`

- [ ] **Step 1: Write `src/Command/SeedCommand.php`**

```php
<?php

declare(strict_types=1);

namespace App\Command;

use Doctrine\DBAL\Connection;
use Symfony\Component\Console\Attribute\AsCommand;
use Symfony\Component\Console\Command\Command;
use Symfony\Component\Console\Input\InputInterface;
use Symfony\Component\Console\Input\InputOption;
use Symfony\Component\Console\Output\OutputInterface;
use Symfony\Component\Console\Style\SymfonyStyle;

#[AsCommand(name: 'app:seed', description: 'Seed the database with test data')]
final class SeedCommand extends Command
{
    public function __construct(private readonly Connection $connection)
    {
        parent::__construct();
    }

    protected function configure(): void
    {
        $this->addOption('users', null, InputOption::VALUE_REQUIRED, 'Number of users', 10)
             ->addOption('accounts', null, InputOption::VALUE_REQUIRED, 'Number of accounts', 20)
             ->addOption('transfers', null, InputOption::VALUE_REQUIRED, 'Number of transfers', 100);
    }

    protected function execute(InputInterface $input, OutputInterface $output): int
    {
        $io        = new SymfonyStyle($input, $output);
        $users     = (int) $input->getOption('users');
        $accounts  = (int) $input->getOption('accounts');
        $transfers = (int) $input->getOption('transfers');

        $io->section("Seeding {$users} users, {$accounts} accounts, {$transfers} transfers");

        $this->connection->beginTransaction();

        try {
            $userIds = $this->seedUsers($users, $io);
            $accountIds = $this->seedAccounts($accounts, $userIds, $io);
            $this->seedTransfers($transfers, $accountIds, $userIds, $io);

            $this->connection->commit();
            $io->success('Seeding complete.');
        } catch (\Throwable $e) {
            $this->connection->rollBack();
            $io->error($e->getMessage());

            return Command::FAILURE;
        }

        return Command::SUCCESS;
    }

    private function seedUsers(int $count, SymfonyStyle $io): array
    {
        $io->writeln("Seeding {$count} users...");
        $ids = [];

        // Bulk insert in batches of 1000
        $batch = [];
        for ($i = 0; $i < $count; $i++) {
            $id     = \App\Domain\User\UserId::generate();
            $ids[]  = (string) $id;
            $batch[] = sprintf(
                "('%s', 'user%d@example.com', 'First%d', 'Last%d', '%s', 'verified', 'active', NOW(), NOW())",
                $id, $i, $i, $i,
                password_hash('Password123!', PASSWORD_ARGON2ID)
            );

            if (count($batch) >= 1000) {
                $this->connection->executeStatement(
                    'INSERT INTO users (id, email, first_name, last_name, password_hash, kyc_status, status, email_verified_at, created_at, updated_at) VALUES ' . implode(',', $batch)
                );
                $batch = [];
            }
        }

        if (!empty($batch)) {
            $this->connection->executeStatement(
                'INSERT INTO users (id, email, first_name, last_name, password_hash, kyc_status, status, email_verified_at, created_at, updated_at) VALUES ' . implode(',', $batch)
            );
        }

        return $ids;
    }

    private function seedAccounts(int $count, array $userIds, SymfonyStyle $io): array
    {
        $io->writeln("Seeding {$count} accounts...");
        $ids       = [];
        $currencies = ['USD', 'EUR', 'GBP', 'CAD', 'AUD'];
        $batch     = [];

        for ($i = 0; $i < $count; $i++) {
            $id       = \App\Domain\Account\AccountId::generate();
            $ids[]    = (string) $id;
            $userId   = $userIds[array_rand($userIds)];
            $currency = $currencies[array_rand($currencies)];

            $batch[] = sprintf(
                "('%s', '%s', '%s', %s, 'Account %d', 'active', NOW(), NOW())",
                $id, $userId, $currency, rand(10000, 1000000) / 100, $i
            );

            if (count($batch) >= 1000) {
                $this->connection->executeStatement(
                    'INSERT INTO accounts (id, user_id, currency, balance, name, status, created_at, updated_at) VALUES ' . implode(',', $batch)
                    . ' ON CONFLICT (user_id, currency) DO NOTHING'
                );
                $batch = [];
            }
        }

        if (!empty($batch)) {
            $this->connection->executeStatement(
                'INSERT INTO accounts (id, user_id, currency, balance, name, status, created_at, updated_at) VALUES ' . implode(',', $batch)
                . ' ON CONFLICT (user_id, currency) DO NOTHING'
            );
        }

        return $ids;
    }

    private function seedTransfers(int $count, array $accountIds, array $userIds, SymfonyStyle $io): void
    {
        $io->writeln("Seeding {$count} transfers...");
        $batch    = [];
        $statuses = ['completed', 'completed', 'completed', 'failed'];

        for ($i = 0; $i < $count; $i++) {
            $fromId   = $accountIds[array_rand($accountIds)];
            $toId     = $accountIds[array_rand($accountIds)];
            if ($fromId === $toId) {
                continue;
            }

            $id       = \App\Domain\Transfer\TransferId::generate();
            $userId   = $userIds[array_rand($userIds)];
            $status   = $statuses[array_rand($statuses)];
            $amount   = rand(100, 10000) / 100;
            $daysAgo  = rand(0, 365);

            $batch[] = sprintf(
                "('%s', '%s', '%s', %s, 'USD', %s, 'USD', %s, 'USD', '1.00000000', 'stub', '%s', 'key-%s', NULL, '%s', '127.0.0.1', 'seed', NOW() - INTERVAL '%d days', NOW() - INTERVAL '%d days')",
                $id, $fromId, $toId, $amount, $amount, $amount, $status, $id, $userId, $daysAgo, $daysAgo
            );

            if (count($batch) >= 1000) {
                $this->connection->executeStatement(
                    'INSERT INTO transfers (id, from_account_id, to_account_id, amount, currency, from_amount, from_currency, to_amount, to_currency, fx_rate, fx_rate_provider, status, idempotency_key, failure_reason, initiated_by_user_id, ip_address, user_agent, created_at, updated_at) VALUES ' . implode(',', $batch)
                    . ' ON CONFLICT DO NOTHING'
                );
                $batch = [];
            }
        }

        if (!empty($batch)) {
            $this->connection->executeStatement(
                'INSERT INTO transfers (id, from_account_id, to_account_id, amount, currency, from_amount, from_currency, to_amount, to_currency, fx_rate, fx_rate_provider, status, idempotency_key, failure_reason, initiated_by_user_id, ip_address, user_agent, created_at, updated_at) VALUES ' . implode(',', $batch)
                . ' ON CONFLICT DO NOTHING'
            );
        }
    }
}
```
